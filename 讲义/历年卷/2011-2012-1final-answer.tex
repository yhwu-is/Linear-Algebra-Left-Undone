\section{2011-2012学年线性代数I（H）期末答案}

\begin{enumerate}
    \item 由特征向量定义 $A\alpha = \lambda \alpha$：

    $A(1,1,1)^\mathrm{T}=(1,a+b+c,d+e+f)^\mathrm{T}=\lambda_1(1,1,1)^\mathrm{T}\implies a+b+c=1,d+e+f=1$.

    $A(1,0,-1)^\mathrm{T}=(0, a-c, d-f)^\mathrm{T}=\lambda_2(1,0,-1)^\mathrm{T}\implies a=c, d=f$.

    $A(1,-1,0)^\mathrm{T}=(2, a-b, d-e)^\mathrm{T}=\lambda_3(1,-1,0)^\mathrm{T}\implies a-b=-2, d=e$.

    综合上述结果可知 $a=c=-\dfrac{1}{3}， b=\dfrac{5}{3}, d=e=f=\dfrac{1}{3}$.

    \item
    \begin{enumerate}
        \item $\alpha \in \ker \sigma_1 \implies \sigma_1(\alpha)=0 \implies \sigma_2(\sigma_1(\alpha))=0 \implies \alpha \in \ker(\sigma_2\circ\sigma_1)$，故 $\ker \sigma_1 \subseteq \ker(\sigma_2\circ\sigma_1)$.
        \item $\beta \in \im(\sigma_2\circ\sigma_1) \implies \exists \alpha,\, \beta = \sigma_2(\sigma_1(\alpha)) \implies \beta \in \im \sigma_2$，故 $\im(\sigma_2\circ\sigma_1) \subseteq \im \sigma_2$.
    \end{enumerate}

    \item
    \begin{enumerate}
        \item $A = \begin{pmatrix} 0 & 0 & 1 \\ 1 & 0 & -1 \\ 1 & 1 & 0 \end{pmatrix}$.
        \item $(v_2,v_3+v_1,v_1-v_2)=(v_1,v_2,v_3)\begin{pmatrix}
            0 & 1 & 1 \\
            1 & 0 & -1 \\
            0 & 1 & 0
        \end{pmatrix}:=(v_1,v_2,v_3)C$，且 $\det(C)=1 \neq 0$，故变换矩阵 $C$ 可逆，$(v_2,v_3+v_1,v_1-v_2)$ 也为 $V$ 的一组基.
        \item 即为 $C^{-1}AC = \begin{pmatrix}
            -1 & 0 & 1 \\
            1 & 1 & 0 \\
            -1 & 0 & 0
        \end{pmatrix}$.
    \end{enumerate}

    \item 由于 $v_1,\ldots,v_n$ 为标准正交基，故 $u = \displaystyle\sum_{i=1}^n \langle u, v_i \rangle v_i$，从而
    \[
        \|u\|^2 = \left\langle \sum_{i=1}^n \langle u, v_i \rangle v_i, \sum_{j=1}^n \langle u, v_j \rangle v_j \right\rangle = \sum_{i=1}^n \sum_{j=1}^n \langle u, v_i \rangle \langle u, v_j \rangle \langle v_i, v_j \rangle = \sum_{i=1}^n \langle u, v_i \rangle^2.
    \]

    \item
    \begin{enumerate}
        \item 根据\autoref{def:内积的定义} 自行验证.
        \item $e_1 = \dfrac{1}{\sqrt{2}}$，$e_2 = \sqrt{\dfrac{3}{2}} x$，$e_3 = \sqrt{\dfrac{5}{8}}(3x^2 - 1)$.
    \end{enumerate}

    \item
    \begin{enumerate}
        \item 同构映射核为 $\{\vec{0}\}$，故无零特征值.
        \item $\sigma(v)=\lambda v \iff v = \lambda \sigma^{-1}(v) \iff \sigma^{-1}(v) = \lambda^{-1}v$.
        \item 由两者特征向量组相同可得.
    \end{enumerate}

    \item
    \begin{enumerate}
        \item \begin{align*}
            f(x_1,x_2,x_3) &= x_1^2+ 6x_2^2 + 4x_3^2 - 4x_1x_2+ 6x_1x_3 - 18x_2x_3 \\
            &= (x_1 - 2x_2 + 3x_3)^2 + 2(x_2 - \dfrac{3}{2}x_3)^2 - \dfrac{19}{2}x_3^2.
        \end{align*}
        因此其相合标准型为 $\diag(1, 2, -\dfrac{19}{2})$，故秩为 $3$，正惯性指数为 $2$，负惯性指数为 $1$.
        \item \begin{align*}
            f(x_1,x_2,x_3,x_4) &= x_1^2 + 2x_2^2 + 10x_3^2 - 14x_4^2 + 2x_1x_2 - 4x_1x_3 \\
            &- 6x_1x_4 - 10x_2x_3 - 2x_2x_4 + 18x_3x_4 \\
            &= (x_1 + x_2 - 2x_3 - 3x_4)^2 + (x_2 - 3x_3 + 2x_4)^2 - 3(x_3 - 3x_4)^2.
        \end{align*}
        因此其相合标准型为 $\diag(1, 1, -3, 0)$，故秩为 $3$，正惯性指数为 $2$，负惯性指数为 $1$.
    \end{enumerate}

    \item
    \begin{enumerate}
        \item 特征多项式相同，且均为对角阵（可对角化），故相似.
        \item 特征值符号（正负个数）相同（重排不改变正负元素个数），故根据惯性定理相合.
    \end{enumerate}

    \item
    \begin{enumerate}
        \item 错. 仅需存在一个向量可由其余表出.
        \item 错. 加法不封闭（$E + (-E) = 0$）.
        \item 对. $A^n=0 \implies$ 特征值全为 0 $\implies |A|=0$.
        \item 对. 总能取 $\varepsilon>0$ 充分小，使得 $1 + \varepsilon \lambda_i > 0$，从而 $E + \varepsilon A$ 正定.
    \end{enumerate}
\end{enumerate}
