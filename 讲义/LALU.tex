% CTeX 会把 chapter 的 label 值设成标题中文编号；这里恢复数字引用语义。
\cs_new_protected:Npn \lalu_chapter_restore_current_label:
  {
    \protected@xdef \@currentlabel
      { \LALUCurrentReferenceNumber }
    \cs_if_exist:NT \cref@constructprefix
      {
        \cref@constructprefix{chapter}{\cref@result}
        \str_if_eq:VnTF \g_lalu_chapter_kind_tl { appendix }
          { \def\lalu@creftype{appendix} }
          { \def\lalu@creftype{chapter} }
        \protected@xdef \cref@currentlabel
          {
            [\lalu@creftype][\arabic{chapter}][\cref@result]
            \LALUCurrentReferenceNumber
          }
      }
  }
\cs_new_protected:Npn \lalu_chapter_clear_current_label:
  {
    \global\let\@currentlabel\@empty
    % LaTeX 2025+ label hooks use this counter name to reconstruct package
    % metadata.  An unnumbered chapter must not inherit the previous chapter.
    \global\let\@currentcounter\@empty
    \cs_if_exist:NT \cref@currentlabel
      { \global\let\cref@currentlabel\@empty }
  }

\cs_new_protected:Npn \lalu_unfinished_chapter:nn #1#2
  {
    \str_if_eq:VnTF \g_lalu_chapter_phase_tl { mainmatter }
      {
        \int_compare:nNnTF { \value{LALUnormalchapter} } > { 0 }
          {
            \lalu_chapter_if_numbered:TF
              {
                \int_compare:nNnTF { \value{LALUunfinishedslot} } < { 6 }
                  {
                    \lalu_chapter_set_unfinished:nn {#1}{#2}
                    \lalu_chapter_call_standard:nn {#1}{#2}
                    \lalu_chapter_restore_current_label:
                  }
                  {
                    \PackageError{LALU}
                      {Too~many~unfinished~topics~after~chapter~\LALUNormalChapterNumber}
                      {Only~six~Greek~slots~are~configured.~Extend~the~central~mapping~before~adding~another~topic.}
                  }
              }
              {
                \PackageError{LALU}
                  {An~unfinished~topic~cannot~be~unnumbered}
                  {Restore~chapter~numbering~before~using~\string\LUchapter.}
              }
          }
          {
            \PackageError{LALU}
              {An~unfinished~topic~requires~a~preceding~normal~chapter}
              {Start~a~numbered~normal~chapter~before~using~\string\LUchapter.}
          }
      }
      {
        \PackageError{LALU}
          {\string\LUchapter~is~only~valid~in~mainmatter}
          {Move~the~unfinished~topic~after~\string\mainmatter~and~a~numbered~normal~chapter.}
      }
  }
